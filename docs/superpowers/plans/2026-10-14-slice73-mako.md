Create `W/lockdelta73.py` with the Write tool, exactly:

```python
"""Slice 73 lock-delta check: main uv.lock vs patched uv.lock (exit 1 on any surprise)."""

import hashlib
import sys
import tomllib

PROOF_SHA256 = "e229c405fdcd27dc371e18f0ba736a193bc5a20072bbee0ab0dbc24c903caa5f"
HOST = "https://files.pythonhosted.org/"

A = tomllib.load(open(sys.argv[1], "rb"))
B = tomllib.load(open(sys.argv[2], "rb"))
ok = bad = 0


def other(msg):
    global bad
    bad += 1
    print("OTHER", msg)


sha = hashlib.sha256(open(sys.argv[2], "rb").read()).hexdigest()
if sha == PROOF_SHA256:
    print("SHA256-OK", sha)
else:
    other(f"sha256 {sha} != proof {PROOF_SHA256}")

top = sorted(k for k in set(A) | set(B) if k != "package" and A.get(k) != B.get(k))
if top:
    other(f"top-level keys changed {top}")


def index(lock):
    return {(p["name"], str(p.get("source"))): p for p in lock.get("package", [])}


a, b = index(A), index(B)
for key in sorted(set(a) | set(b)):
    pa, pb = a.get(key), b.get(key)
    if pa == pb:
        continue
    name = key[0]
    if name != "mako" or pa is None or pb is None:
        other(f"package {'added' if pa is None else 'removed' if pb is None else 'changed'} {name!r}")
        continue
    changed = sorted(k for k in set(pa) | set(pb) if pa.get(k) != pb.get(k))
    if (pa["version"], pb["version"]) != ("1.4.1", "1.4.2"):
        other(f"mako version {pa['version']} -> {pb['version']}")
    elif not set(changed) <= {"version", "sdist", "wheels"}:
        other(f"mako fields changed {changed}")
    else:
        urls = [pb["sdist"]["url"]] + [w["url"] for w in pb.get("wheels", [])]
        good = all(u.startswith(HOST) and "mako-1.4.2" in u for u in urls)
        if good:
            ok += 1
            print("OK    changed 'mako' 1.4.1 -> 1.4.2", changed)
        else:
            other(f"mako url not {HOST} or not 1.4.2: {urls}")

if ok != 1:
    other(f"expected 1 OK row, got {ok}")
print(f"ok={ok} other={bad}")
sys.exit(1 if bad else 0)
```
