PROOF_SHA256 = "9702766ff08ab3b34fbcc3657a0e0c8ce499821dc96766e1094b793aef1e6f99"

a = json.load(open(sys.argv[1]))["packages"]
b = json.load(open(sys.argv[2]))["packages"]
ok = bad = 0

if sha == PROOF_SHA256:
    print("SHA256-OK", sha)
else:
    bad += 1
    print("OTHER sha256", sha, "!= proof", PROOF_SHA256)


def row(good, kind, key, detail=""):
    global ok, bad
    if good:
        ok += 1
    else:
        bad += 1
    print("OK   " if good else "OTHER", kind, repr(key), detail)


for k in sorted(set(a) | set(b)):
    va, vb = a.get(k), b.get(k)
    if va == vb:
        if k in FROZEN:
            print("FROZEN-OK", repr(k), va.get("version"))
        continue
    if k == "":
        da, db = va.get("devDependencies", {}), vb.get("devDependencies", {})
        rest_same = {x: y for x, y in va.items() if x != "devDependencies"} == {
            x: y for x, y in vb.items() if x != "devDependencies"
        }
        keys = {x for x in set(da) | set(db) if da.get(x) != db.get(x)}
        good = rest_same and keys == set(ROOT_DEV) and all(
            (da.get(x), db.get(x)) == ROOT_DEV[x] for x in ROOT_DEV
        )
        row(good, "root", k, sorted(keys))
    elif k in FROZEN:
        row(False, "frozen-changed", k)
    elif "tinypool" in k and vb is not None:
        row(False, "tinypool-present", k)
    elif vb is None:
        row(REMOVED.get(k) == va.get("version"), "removed", k, va.get("version"))
    elif va is None:
        row(ADDED.get(k) == vb.get("version"), "added", k, vb.get("version"))
    else:
        vers = (va.get("version"), vb.get("version"))
        same_ver = vers[0] == vers[1]
        row(not same_ver and CHANGED.get(k) == vers, "changed", k, vers)

if any("tinypool" in k for k in b):
    bad += 1

expected = 1 + len(ADDED) + len(REMOVED) + len(CHANGED)
if ok != expected:
    bad += 1
    print(f"OTHER expected {expected} OK rows, got {ok}")
print(f"ok={ok} other={bad}")
sys.exit(1 if bad else 0)
