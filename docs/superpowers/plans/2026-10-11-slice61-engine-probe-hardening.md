def _recorder(ledger):
    """A spend.record stand-in that appends to the ledger and closes it."""

    def record(kind, micros, release=0):
        with ledger.open("a") as f:
            f.write(json.dumps({"kind": kind, "micros": micros}) + "\n")

    return record
